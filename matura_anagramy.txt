#include <iostream>
#include <fstream>
#include <string>
#include <vector>
#include <set>

using namespace std;

int licz_jedynki(string tekst){
    int licznik = 0;
    for(char h: tekst){
        if(h == '1'){
            licznik++;
        }
    }
    return licznik;
}

int dwojki_na_dzies(string l){ 
    int liczba = 0; 
    int waga = l.size() - 2; // znak bialy na koncu linii
    for(int i = 0; i < l.size(); i++){ 
        if(l[i] == '1'){ 
            int potega = 1;
            for(int j = 0; j < waga - i; j++){ 
                potega *= 2; 
            } 
            liczba += potega; 
        }
        
    } 
    return liczba; 
}

string dzies_na_dwojki(int li){
    string binarna;
    while(li > 0){
        if(li%2 == 0){
            binarna = "0"+binarna;
        }
        else{
            binarna = "1"+binarna;
        }
        li /= 2;
    }
    
    return binarna;
}

int wartosc_bezwzgledna(int liczba){
    if(liczba < 0){
        liczba *= -1;
    }
    return liczba;
}

void licz_jedynki_i_zera(){
    int zrownowazoneLiczby = 0;
    int prawieZrownowazone = 0;

    ifstream wejdzPlik("przyklad.txt");
    ofstream zapisWynikuPlik("wyniki3.txt");

    if(!wejdzPlik.is_open()){
        cerr << "Blad przy otwieraniu pliku anagram.txt" << endl;
        return;
    }

    if(!zapisWynikuPlik.is_open()){
        cerr << "Blad, plik wyniki3.txt nie moze zostac otwarty" << endl;
        return;
    }

    string linia;

    while(getline(wejdzPlik, linia)){
        int licznikZera = 0;
        int licznikJedynki = 0;

        for(char c : linia){
            if(c == '0'){
                licznikZera++;
            }
            else if(c == '1'){
                licznikJedynki++;
            }
        }

        if(licznikJedynki == licznikZera){
            zrownowazoneLiczby++;
        } 
        else if(wartosc_bezwzgledna(licznikZera - licznikJedynki) == 1){
            prawieZrownowazone++;
        }
    }

    zapisWynikuPlik<<"Zadanie 3.1."<<endl;
    zapisWynikuPlik<<"Liczby zrownowazone: "<<zrownowazoneLiczby<<endl;
    zapisWynikuPlik<<"Liczby prawie zrównowazone: "<<prawieZrownowazone<<endl;

    cout<<"Ilosc zrownowazonych: "<<zrownowazoneLiczby<<endl;
    cout<<"Ilosc prawie zrownowazonych: "<<prawieZrownowazone<<endl;

    wejdzPlik.close();
    zapisWynikuPlik.close();
}

void anagram_cyfr(){
    vector<string> liczby;
    ifstream wejdzPlik("przyklad.txt");
    ofstream zapisWynikuPlik("wyniki3.txt");
    
    string linia;
    while(getline(wejdzPlik, linia)){
        //cout<<(linia.size()-1 == 8 ? linia : "")<<endl;
        liczby.push_back(linia);
    }
    
    for(int i = 0; i<liczby.size(); i++){
        if(liczby[i].length()-1 == 8 && licz_jedynki(liczby[i]) > 3 && licz_jedynki(liczby[i]) < 6){
            cout<<liczby[i]<<endl;
        }
    }
    
    wejdzPlik.close();
    zapisWynikuPlik.close();
}

void sasiedzi_liczbowi(){
    vector<int> liczby_dziesietne;
    string linia;
    int roznica = 0, najwRoznica = 0;
    ifstream wejdzPlik("anagram.txt");
    ofstream zapisWynikuPlik("wyniki3.txt");
    
    while(getline(wejdzPlik, linia)){
        liczby_dziesietne.push_back(dwojki_na_dzies(linia));
    }
    
    for(size_t it = 0; it<liczby_dziesietne.size()-1; it++){
        roznica = wartosc_bezwzgledna(liczby_dziesietne[it] - liczby_dziesietne[it+1]);
        if(roznica>najwRoznica){
            najwRoznica = roznica;
        }
    }
    
    string najwRozBin = dzies_na_dwojki(najwRoznica);

    zapisWynikuPlik<<"Najwieksza roznica (bin): "<<najwRozBin;
    
    wejdzPlik.close();
    zapisWynikuPlik.close();
}

void suma_roznych(){
    set<int> unikal_cyfry;
    
    int liczba_dz = 0;
    int najwLiczbDz = 0;
    int bezCyfryZero = 0;
    int sumaRoz = 0, najwSuma = 0;
    ifstream wejdzPlik("anagram.txt");
    ofstream zapisWynikuPlik("wyniki3.txt");
    
    string linia;
    while(getline(wejdzPlik, linia)){
        sumaRoz = 0;
        liczba_dz = (dwojki_na_dzies(linia));
        string l_str = to_string(liczba_dz);
        
        bool maZero = false; 
        for(char zer: l_str){ 
            if(zer == '0'){ 
                maZero = true; 
                break; 
            } 
        } 
        if(!maZero){ 
            bezCyfryZero++; 
        }
        
        for(char roz: l_str){
            unikal_cyfry.insert(roz);
        }
        
        for(int sRoz: unikal_cyfry){
            sumaRoz += sRoz;
        }
        
        if(sumaRoz > najwSuma){
            najwSuma = sumaRoz;
            najwLiczbDz = liczba_dz;
        }
        
        unikal_cyfry.clear();
    }

    cout<<"Ilosc liczb bez zera: "<<bezCyfryZero<<endl;
    cout<<"Najwieksza suma roznych cyfr (liczba): "<<najwLiczbDz<<endl;
}

int main()
{
    licz_jedynki_i_zera();
    anagram_cyfr();
    sasiedzi_liczbowi();
    suma_roznych();
    return 0;
}
